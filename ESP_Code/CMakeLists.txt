cmake_minimum_required(VERSION 3.16)
project(esp_motor_core LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(motor_core src/controller.cpp src/motor.cpp src/encoder.cpp
    src/speed_control.cpp src/communication.cpp)
target_include_directories(motor_core PUBLIC include)
if(MSVC)
    target_compile_options(motor_core PRIVATE /W4 /WX /utf-8)
else()
    target_compile_options(motor_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_executable(test_core tests/test_core.cpp)
target_link_libraries(test_core PRIVATE motor_core)
add_executable(native_sim tools/native_sim.cpp)
target_link_libraries(native_sim PRIVATE motor_core)
add_executable(motor_duty_example examples/motor_duty_example.cpp)
target_link_libraries(motor_duty_example PRIVATE motor_core)
enable_testing()
add_test(NAME core COMMAND test_core)
add_test(NAME example COMMAND motor_duty_example)
